-- 1. Debarment Check (HAVING + Derived Table Subquery)
SELECT s.Person_ID AS Student_ID, p.Full_Name, s.Enrollment_No,
     c.Course_Code, c.Course_Name, sec.Section_Name,
     COUNT(a.Session_ID) AS Total_Sessions,
     SUM(CASE WHEN a.Status IN ('Present', 'Late') THEN 1 ELSE 0 END) AS Present_Count,
     ROUND(100.0 * SUM(CASE WHEN a.Status IN ('Present', 'Late') THEN 1 ELSE 0 END)
         / NULLIF(COUNT(a.Session_ID), 0), 2) AS Attendance_Pct
FROM ENROLLMENT e
JOIN STUDENT s ON e.Student_ID = s.Person_ID
JOIN PERSON p ON s.Person_ID = p.Person_ID
JOIN COURSE c ON e.Course_ID = c.Course_ID
JOIN SECTION sec ON e.Section_ID = sec.Section_ID
LEFT JOIN (
    SELECT a2.Student_ID, a2.Session_ID, a2.Status, t.Course_ID, t.Section_ID
    FROM ATTENDANCE a2
    JOIN ATTENDANCE_SESSION as2 ON a2.Session_ID = as2.Session_ID
    JOIN TIMETABLE t ON as2.Timetable_ID = t.Timetable_ID
) a ON a.Student_ID = e.Student_ID AND a.Course_ID = e.Course_ID AND a.Section_ID = e.Section_ID
GROUP BY s.Person_ID, p.Full_Name, s.Enrollment_No, c.Course_Code, c.Course_Name, sec.Section_Name
HAVING Attendance_Pct < 75 AND Total_Sessions > 0
ORDER BY Attendance_Pct ASC;
-- 2. Global Audit Trail (Multi-Table JOIN, No Faculty Filter)
SELECT aa.Audit_ID, p.Full_Name AS Student_Name,
     aa.Session_ID, DATE_FORMAT(ats.Session_Date, '%Y-%m-%d') AS Session_Date,
     c.Course_Code, aa.Old_Status, aa.New_Status,
     aa.Changed_By, CAST(aa.Changed_At AS CHAR) AS Changed_At, aa.Change_Reason
FROM ATTENDANCE_AUDIT aa
JOIN PERSON p ON aa.Student_ID = p.Person_ID
JOIN ATTENDANCE_SESSION ats ON aa.Session_ID = ats.Session_ID
JOIN TIMETABLE t ON ats.Timetable_ID = t.Timetable_ID
JOIN COURSE c ON t.Course_ID = c.Course_ID
ORDER BY aa.Audit_ID DESC;

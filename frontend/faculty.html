-- 3. Strict Role-Isolated Timetable Query
SELECT t.Timetable_ID, c.Course_Name, s.Section_Name, r.Room_Number, ts.Day_Of_Week, ts.Start_Time, ts.End_Time
FROM TIMETABLE t
JOIN COURSE c ON t.Course_ID = c.Course_ID
JOIN SECTION s ON t.Section_ID = s.Section_ID
JOIN ROOM r ON t.Room_ID = r.Room_ID
JOIN TIME_SLOT ts ON t.Slot_ID = ts.Slot_ID
WHERE t.Faculty_ID = 1;

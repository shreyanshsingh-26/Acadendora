-- 1. Student Profile & Enrolled Batch Resolution
SELECT p.Full_Name, s.Enrollment_No, s.Status, d.Dept_Name, sec.Section_Name AS Batch_Name
FROM student s
JOIN person p ON s.Person_ID = p.Person_ID
JOIN department d ON p.Department_ID = d.Department_ID
LEFT JOIN enrollment e ON s.Person_ID = e.Student_ID
LEFT JOIN section sec ON e.Section_ID = sec.Section_ID
WHERE s.Person_ID = 10
LIMIT 1;
-- 2. Pre-Computed Summary View (Institutional Performance)
SELECT Course_Code, Course_Name, Total_Sessions, Present_Count, Absent_Count, Attendance_Pct
FROM Student_Attendance_Summary
WHERE Full_Name = (SELECT Full_Name FROM person WHERE Person_ID = 10);
-- 3. 6-Table Timetable Schedule Join
SELECT c.Course_Name, c.Course_Code, p.Full_Name AS Faculty_Name, r.Room_Number, ts.Day_Of_Week, ts.Start_Time, ts.End_Time
FROM timetable t
JOIN course c ON t.Course_ID = c.Course_ID
JOIN faculty f ON t.Faculty_ID = f.Person_ID
JOIN person p ON f.Person_ID = p.Person_ID
JOIN room r ON t.Room_ID = r.Room_ID
JOIN time_slot ts ON t.Slot_ID = ts.Slot_ID
JOIN enrollment e ON t.Section_ID = e.Section_ID
WHERE e.Student_ID = 10
ORDER BY ts.Slot_ID;

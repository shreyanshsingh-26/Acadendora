-- 1. Profile Resolution:
SELECT p.First_Name, p.Last_Name, s.Enrollment_No, b.Batch_Name, d.Department_Name
FROM STUDENT s JOIN PERSON p ON s.Person_ID = p.Person_ID
JOIN BATCH b ON s.Batch_ID = b.Batch_ID JOIN DEPARTMENT d ON s.Department_ID = d.Department_ID
WHERE s.Student_ID = 10;

-- 3. Timetable Schedule (6-Table Join):
SELECT c.Course_Name, p.First_Name, r.Room_Number, ts.Day_Of_Week, ts.Start_Time
FROM TIMETABLE t JOIN COURSE c ON t.Course_ID = c.Course_ID
JOIN FACULTY f ON t.Faculty_ID = f.Faculty_ID JOIN PERSON p ON f.Person_ID = p.Person_ID
JOIN ROOM r ON t.Room_ID = r.Room_ID JOIN TIME_SLOT ts ON t.Slot_ID = ts.Slot_ID
WHERE t.Section_ID = 1 ORDER BY ts.Slot_ID;
